# Run from the repository root. Local image and container are named mole-game.
docker build -t mole-game:latest ./python

# Replace the device path with the cabinet Arduino's stable host path.
docker run -d \
  --name mole-game \
  --restart unless-stopped \
  --device=/dev/serial/by-id/REPLACE_WITH_ARDUINO:/dev/cabinet-arduino \
  -e ARDUINO_PORT=/dev/cabinet-arduino \
  -e STATUS_PORT=8080 \
  --mount type=volume,source=mole-game-data,target=/data \
  -p 8080:8080 \
  --log-driver=json-file \
  --log-opt max-size=10m \
  --log-opt max-file=3 \
  mole-game:latest

docker logs --tail 100 -f mole-game
docker inspect --format '{{.State.Health.Status}}' mole-game
docker stop --time 15 mole-game
